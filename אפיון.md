import { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { scheduleAPI } from '../api/scheduleAPI';
import {
  Box,
  Button,
  Container,
  Paper,
  Typography,
  Select,
  MenuItem,
  FormControl,
  InputLabel,
  Table,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  CircularProgress,
} from '@mui/material';
import { BrowserRouter, Routes, Route } from 'react-router-dom';
import { ScheduleProvider } from './context/ScheduleContext';
import LoadFilePage from './pages/LoadFilePage';
import SelectViewPage from './pages/SelectViewPage';
import ClassSchedulePage from './pages/ClassSchedulePage';
import TeacherSchedulePage from './pages/TeacherSchedulePage';
import DaySchedulePage from './pages/DaySchedulePage';
import Navbar from './components/Navbar';
import axios from 'axios';

const DAYS = ['ראשון', 'שני', 'שלישי', 'רביעי', 'חמישי', 'שישי'];

export default function DaySchedulePage() {
  const [selectedDay, setSelectedDay] = useState('ראשון');
  const [schedule, setSchedule] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const navigate = useNavigate();

  useEffect(() => {
    fetchDaySchedule(selectedDay);
  }, [selectedDay]);

  useEffect(() => {
    checkScheduleStatus();
  }, []);

  const checkScheduleStatus = async () => {
    try {
      const response = await axios.get('http://localhost:5000/api/schedule/status');
      if (response.data.isLoaded) {
        // Schedule already loaded, redirect to select page
        navigate('/select');
      }
    } catch (error) {
      // Server not running or no schedule loaded
    }
  };

  const fetchDaySchedule = async (day) => {
    setLoading(true);
    setError('');
    try {
      const response = await scheduleAPI.getDaySchedule(day);
      setSchedule(response.data);
    } catch (err) {
      setError('שגיאה בטעינת מערכת היום');
    } finally {
      setLoading(false);
    }
  };

  const groupByHour = (lessons) => {
    const grouped = {};
    lessons.forEach(lesson => {
      if (!grouped[lesson.hour]) {
        grouped[lesson.hour] = [];
      }
      grouped[lesson.hour].push(lesson);
    });
    return grouped;
  };

  const groupedSchedule = groupByHour(schedule);

  return (
    <Container maxWidth="lg">
      <Box sx={{ my: 4 }}>
        <Typography variant="h4" component="h1" gutterBottom>
          מערכת לפי יום
        </Typography>

        <FormControl fullWidth sx={{ mb: 3 }}>
          <InputLabel>בחר יום</InputLabel>
          <Select
            value={selectedDay}
            onChange={(e) => setSelectedDay(e.target.value)}
            label="בחר יום"
          >
            {DAYS.map(day => (
              <MenuItem key={day} value={day}>{day}</MenuItem>
            ))}
          </Select>
        </FormControl>

        {loading && <CircularProgress />}
        {error && <Typography color="error">{error}</Typography>}

        {!loading && !error && (
          <TableContainer component={Paper}>
            <Table>
              <TableHead>
                <TableRow>
                  <TableCell>שעה</TableCell>
                  <TableCell>כיתה</TableCell>
                  <TableCell>מקצוע</TableCell>
                  <TableCell>מורה</TableCell>
                  <TableCell>הקבצה</TableCell>
                </TableRow>
              </TableHead>
              <TableBody>
                {Object.entries(groupedSchedule).map(([hour, lessons]) => (
                  <React.Fragment key={hour}>
                    {lessons.map((lesson, index) => (
                      <TableRow key={`${hour}-${index}`}>
                        {index === 0 && (
                          <TableCell rowSpan={lessons.length}>
                            {hour}
                          </TableCell>
                        )}
                        <TableCell>{lesson.className}</TableCell>
                        <TableCell>{lesson.subject}</TableCell>
                        <TableCell>{lesson.teacher}</TableCell>
                        <TableCell>{lesson.group || '-'}</TableCell>
                      </TableRow>
                    ))}
                  </React.Fragment>
                ))}
              </TableBody>
            </Table>
          </TableContainer>
        )}

        <Box sx={{ mt: 3 }}>
          <Button variant="outlined" onClick={() => navigate('/select')}>
            חזרה לתפריט
          </Button>
        </Box>
      </Box>
    </Container>
  );
}

function App() {
  return (
    <BrowserRouter>
      <ScheduleProvider>
        <Navbar />
        <Routes>
          <Route path="/" element={<LoadFilePage />} />
          <Route path="/select" element={<SelectViewPage />} />
          <Route path="/schedule/class" element={<ClassSchedulePage />} />
          <Route path="/schedule/teacher" element={<TeacherSchedulePage />} />
          <Route path="/schedule/day" element={<DaySchedulePage />} />
        </Routes>
      </ScheduleProvider>
    </BrowserRouter>
  );
}

export default App;

import { AppBar, Toolbar, Button, Typography } from '@mui/material';

export default function Navbar() {
  const navigate = useNavigate();

  return (
    <AppBar position="static">
      <Toolbar>
        <Typography variant="h6" component="div" sx={{ flexGrow: 1 }}>
          מערכת תזמון בית ספרית
        </Typography>
        <Button color="inherit" onClick={() => navigate('/')}>
          טעינת קובץ
        </Button>
        <Button color="inherit" onClick={() => navigate('/select')}>
          תפריט ראשי
        </Button>
      </Toolbar>
    </AppBar>
  );
}
